cmake_minimum_required(VERSION 3.24)
project(MizinPlayout LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Concurrent)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../evidence/build-version.json")
  message(FATAL_ERROR "Use scripts/build.ps1 to generate the version from target commits first.")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../evidence/build-version.json" VERSION_JSON)
string(JSON MIZIN_VERSION GET "${VERSION_JSON}" display)
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
qt_add_executable(MizinPlayout src/main.cpp src/channel.cpp src/channel.hpp src/cue.hpp src/compositor.hpp src/playlist.hpp src/smoke.cpp src/smoke.hpp src/transport_tests.cpp)
target_include_directories(MizinPlayout PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(MizinPlayout PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Concurrent)
if(MSVC)
  target_compile_options(MizinPlayout PRIVATE /utf-8 /W4)
endif()
set_target_properties(MizinPlayout PROPERTIES WIN32_EXECUTABLE TRUE)
qt_add_resources(MizinPlayout "product" PREFIX "/" FILES i18n/en.json i18n/pt-BR.json)
